from fastapi import FastAPI, HTTPException
import sqlite3

app = FastAPI(title="National Weather Analytics API")


DATABASE = r"C:\Users\Vidhyarth Nagar\Desktop\Hackathon\weather_platform.db"


def get_db():
    conn = sqlite3.connect(DATABASE)
    conn.row_factory = sqlite3.Row
    return conn


@app.get("/")
def home():
    return {
        "message": "National Weather Analytics API is running"
    }


@app.get("/reports")
def get_reports(
    city: str | None = None,
    source: str | None = None,
    event_category: str | None = None
):
    conn = get_db()

    query = "SELECT * FROM weather_events WHERE 1=1"
    params = []

    if city:
        query += " AND city = ?"
        params.append(city)

    if source:
        query += " AND source = ?"
        params.append(source)

    if event_category:
        query += " AND event_category = ?"
        params.append(event_category)

    query += " ORDER BY rowid DESC"

    rows = conn.execute(query, params).fetchall()

    conn.close()

    return {
        "count": len(rows),
        "reports": [dict(row) for row in rows]
    }


@app.get("/reports/{event_id}")
def get_report(event_id: str):
    conn = get_db()

    row = conn.execute(
        "SELECT * FROM weather_events WHERE event_id = ?",
        (event_id,)
    ).fetchone()

    conn.close()

    if row is None:
        raise HTTPException(
            status_code=404,
            detail="Weather report not found"
        )

    return dict(row)
